<script>
const nuinLogo = new Image();
nuinLogo.src = "white_logo_transparent_background.png";

const canvas = document.getElementById("stakeholdersCanvas");
if (canvas) {
  const ctx = canvas.getContext("2d");
  const rootStyles = getComputedStyle(document.documentElement);
  const GREEN = (rootStyles.getPropertyValue("--green-light") || "#00C98E").trim();
  const DEEP_GREEN = (rootStyles.getPropertyValue("--green-dark") || "#007A58").trim();

  let width = 0;
  let height = 0;
  let DPR = window.devicePixelRatio || 1;
  let centerX = 0;
  let centerY = 0;
  let mouse = { x: -1000, y: -1000 };
  let nodes = [];
  let pulses = [];
  let orbitX = 0;
  let orbitY = 0;
  let maxOuterHalfW = 0;
  let maxOuterHalfH = 0;

  const THEME = {
    green: GREEN,
    deepGreen: DEEP_GREEN,
    text: "#ffffff",
    centerText: "#010807"
  };

  const nodeData = [
    { title: "KOLs", subtitle: "Líderes de Opinión" },
    { title: "Farmacia Hospitalaria", subtitle: "Entorno Hospitalario" },
    { title: "Farmacia de Calle", subtitle: "Retail & Community" },
    { title: "Cuidadores", subtitle: "Entorno del Paciente" },
    { title: "Asoc. Pacientes", subtitle: "Asociaciones" },
    { title: "Pacientes", subtitle: "La Voz Real" },
    { title: "Enfermería", subtitle: "Cuidado Profesional" },
    { title: "Laboratorios", subtitle: "Industria Farmacéutica" },
    { title: "Hospitales", subtitle: "Instituciones de Salud" },
    { title: "HCPs", subtitle: "Profesionales Médicos" }
  ];

  function isMobile() { return width < 768; }

  function capsulePath(x, y, w, h) {
    const r = h / 2;
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }

  class Node {
    constructor(id, title, subtitle, isCenter = false) {
      this.id = id;
      this.title = title;
      this.subtitle = subtitle;
      this.isCenter = isCenter;
      this.angle = isCenter ? 0 : ((id - 1) * (Math.PI * 2) / nodeData.length);
      this.x = 0;
      this.y = 0;
      this.h = isCenter ? 110 : 68;
      this.paddingSide = isCenter ? 80 : 36;
      this.w = 0;
      this.floatOffset = Math.random() * Math.PI * 2;
      this.hoverProgress = 0;
      this.currentW = 0;
      this.currentH = 0;
    }

    calculateSize() {
      const titleSize = this.isCenter ? 40 : 14;
      ctx.font = `600 ${titleSize}px Montserrat, system-ui, sans-serif`;
      const titleW = ctx.measureText(this.title || "").width;

      const subSize = this.isCenter ? 12 : 9;
      ctx.font = `500 ${subSize}px Montserrat, system-ui, sans-serif`;
      const subW = this.subtitle ? ctx.measureText(this.subtitle.toUpperCase()).width : 0;

      this.w = Math.max(titleW, subW) + this.paddingSide;
      this.w = this.isCenter ? Math.max(this.w, 280) : Math.max(this.w, 120);
    }

    update(time) {
      const floatSpeed = 0.5;
      const floatY = Math.sin(time * floatSpeed + this.floatOffset) * 4;
      const floatX = Math.cos(time * floatSpeed * 0.8 + this.floatOffset) * 2;

      if (this.isCenter) {
        this.x = centerX + floatX;
        this.y = centerY + floatY;
      } else {
        const targetX = centerX + Math.cos(this.angle) * orbitX;
        const targetY = centerY + Math.sin(this.angle) * orbitY;
        this.x = targetX + floatX;
        this.y = targetY + floatY;
      }

      const dx = mouse.x - this.x;
      const dy = mouse.y - this.y;
      const hover = Math.abs(dx) < this.currentW / 2 && Math.abs(dy) < this.currentH / 2;
      this.hoverProgress = hover ? Math.min(1, this.hoverProgress + 0.15) : Math.max(0, this.hoverProgress - 0.1);

      this.currentW = this.w + (this.hoverProgress * 20);
      this.currentH = this.h + (this.hoverProgress * 4);
    }

    draw() {
      ctx.save();
      ctx.translate(this.x, this.y);

      const w = this.currentW;
      const h = this.currentH;

      ctx.shadowBlur = this.isCenter ? (60 + this.hoverProgress * 30) : (22 + this.hoverProgress * 10);
      ctx.shadowColor = `rgba(0, 201, 142, ${this.isCenter ? 0.8 : 0.45})`;

      const grad = ctx.createLinearGradient(-w / 2, -h / 2, w / 2, h / 2);
      if (this.isCenter) {
        grad.addColorStop(0, THEME.green);
        grad.addColorStop(1, THEME.deepGreen);
      } else {
        grad.addColorStop(0, `rgba(0, 201, 142, ${0.16 + this.hoverProgress * 0.1})`);
        grad.addColorStop(1, `rgba(0, 35, 30, 0.55)`);
      }

      capsulePath(-w / 2, -h / 2, w, h);
      ctx.fillStyle = grad;
      ctx.fill();

      ctx.shadowBlur = 0;
      ctx.strokeStyle = this.isCenter ? "rgba(255,255,255,0.80)" : `rgba(0, 201, 142, ${0.5 + this.hoverProgress * 0.5})`;
      ctx.lineWidth = this.isCenter ? 6 : 1.7;
      ctx.stroke();

      ctx.textAlign = "center";
      ctx.textBaseline = "middle";

      if (this.isCenter) {
        if (nuinLogo.complete && nuinLogo.naturalWidth > 0) {
          const logoW = isMobile() ? 90 : 110;
          const logoH = logoW * 0.35;
          ctx.drawImage(nuinLogo, -logoW / 2, -logoH / 2, logoW, logoH);
        }
      } else {
        ctx.fillStyle = THEME.text;
        ctx.font = `600 ${isMobile() ? '12px' : '14px'} Montserrat, system-ui, sans-serif`;
        ctx.fillText(this.title, 0, this.subtitle ? -10 : 0);

        if (this.subtitle) {
          ctx.fillStyle = `rgba(0, 201, 142, 0.9)`;
          ctx.font = `500 9px Montserrat, system-ui, sans-serif`;
          ctx.fillText(this.subtitle.toUpperCase(), 0, 13);
        }
      }

      ctx.restore();
    }
  }

  class Pulse {
    constructor(target) {
      this.target = target;
      this.reset();
    }
    reset() {
      this.progress = 0;
      this.speed = 0.0025 + Math.random() * 0.003;
    }
    update() {
      this.progress += this.speed;
      if (this.progress >= 1) this.reset();
    }
    draw(center) {
      const x = center.x + (this.target.x - center.x) * this.progress;
      const y = center.y + (this.target.y - center.y) * this.progress;
      const alpha = Math.sin(this.progress * Math.PI);
      ctx.fillStyle = `rgba(0, 201, 142, ${alpha * 1.0})`;
      ctx.shadowBlur = 10;
      ctx.shadowColor = THEME.green;
      ctx.beginPath();
      ctx.arc(x, y, 2.5, 0, Math.PI * 2);
      ctx.fill();
      ctx.shadowBlur = 0;
    }
  }

  function init() {
    nodes = [];
    pulses = [];
    maxOuterHalfW = 0;
    maxOuterHalfH = 0;

    const centerNode = new Node(0, "", "", true);
    centerNode.calculateSize();
    nodes.push(centerNode);

    nodeData.forEach((data, i) => {
      const n = new Node(i + 1, data.title, data.subtitle, false);
      n.calculateSize();
      nodes.push(n);
      pulses.push(new Pulse(n));
      pulses.push(new Pulse(n));

      maxOuterHalfW = Math.max(maxOuterHalfW, (n.w + 20) / 2);
      maxOuterHalfH = Math.max(maxOuterHalfH, (n.h + 4) / 2);
    });
  }

  function resize() {
    DPR = window.devicePixelRatio || 1;
    const rect = canvas.getBoundingClientRect();
    width = rect.width;
    height = Math.max(480, rect.width * 0.70);

    canvas.width = Math.round(width * DPR);
    canvas.height = Math.round(height * DPR);
    canvas.style.height = `${height}px`;

    centerX = width / 2;
    centerY = height / 2;

    init();
  }

  function updateOrbits() {
    const pad = isMobile() ? 18 : 28;
    const availX = Math.max(140, width / 2 - maxOuterHalfW - pad);
    const availY = Math.max(120, height / 2 - maxOuterHalfH - pad);
    const baseX = isMobile() ? 175 : 520;
    const baseY = isMobile() ? 155 : 370;
    orbitX = Math.min(baseX, availX * 0.98);
    orbitY = Math.min(baseY, availY * 0.95, orbitX / 1.4);
  }

  function animate(time) {
    const t = time * 0.001;
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
    ctx.clearRect(0, 0, width, height);

    ctx.fillStyle = "rgba(0, 201, 142, 0.04)";
    for (let i = 0; i < 30; i++) {
      const x = (Math.sin(t * 0.04 + i * 17) * 0.5 + 0.5) * width;
      const y = (Math.cos(t * 0.07 + i * 11) * 0.5 + 0.5) * height;
      ctx.beginPath();
      ctx.arc(x, y, 1, 0, Math.PI * 2);
      ctx.fill();
    }

    updateOrbits();
    nodes.forEach(n => n.update(t));
    pulses.forEach(p => p.update());

    const center = nodes[0];
    nodes.forEach((node, i) => {
      if (i === 0) return;
      ctx.beginPath();
      ctx.lineWidth = 0.8;
      const lineGrad = ctx.createLinearGradient(center.x, center.y, node.x, node.y);
      lineGrad.addColorStop(0.1, "rgba(0, 201, 142, 0.6)");
      lineGrad.addColorStop(1, "rgba(0, 201, 142, 0)");
      ctx.strokeStyle = lineGrad;
      ctx.moveTo(center.x, center.y);
      ctx.lineTo(node.x, node.y);
      ctx.stroke();
    });

    pulses.forEach(p => p.draw(center));
    nodes.forEach(n => n.draw());

    requestAnimationFrame(animate);
  }

  resize();
  window.addEventListener("resize", resize);
  canvas.addEventListener("mousemove", e => {
    const rect = canvas.getBoundingClientRect();
    mouse.x = e.clientX - rect.left;
    mouse.y = e.clientY - rect.top;
  });
  canvas.addEventListener("mouseleave", () => { mouse.x = -1000; mouse.y = -1000; });
  canvas.addEventListener("touchstart", e => {
    const rect = canvas.getBoundingClientRect();
    mouse.x = e.touches[0].clientX - rect.left;
    mouse.y = e.touches[0].clientY - rect.top;
  }, { passive: true });
  canvas.addEventListener("touchmove", e => {
    const rect = canvas.getBoundingClientRect();
    mouse.x = e.touches[0].clientX - rect.left;
    mouse.y = e.touches[0].clientY - rect.top;
  }, { passive: true });
  canvas.addEventListener("touchend", () => { mouse.x = -1000; mouse.y = -1000; }, { passive: true });

  if (nuinLogo.complete) {
    requestAnimationFrame(animate);
  } else {
    nuinLogo.addEventListener("load", () => requestAnimationFrame(animate), { once: true });
  }
}

  const isOpen = panel.classList.contains('open');
  document.querySelectorAll('.acc-panel').forEach(p => p.classList.remove('open'));
  document.querySelectorAll('.xp-acc-btn').forEach(b => b.classList.remove('acc-active'));

  if (!isOpen) {
    panel.classList.add('open');
    if (btn) btn.classList.add('acc-active');
    if (key === 'stakeholders') {
      requestAnimationFrame(() => requestAnimationFrame(() => window.dispatchEvent(new Event('resize'))));
    }
  }
}

function toggleAccordion(key, btn) {
  const idxByKey = { stakeholders: 0, areas: 1, principios: 2 };
  showExperience(idxByKey[key], btn);
}

function setSalaTab(key) {
  const isNuin = key === 'nuin';
  const tabNuin = document.getElementById('sala-tab-nuin');
  const tabFocus = document.getElementById('sala-tab-focus');
  const panelNuin = document.getElementById('sala-panel-nuin');
  const panelFocus = document.getElementById('sala-panel-focus');
  if (tabNuin) tabNuin.classList.toggle('active', isNuin);
  if (tabFocus) tabFocus.classList.toggle('active', !isNuin);
  if (panelNuin) panelNuin.classList.toggle('active', isNuin);
  if (panelFocus) panelFocus.classList.toggle('active', !isNuin);
}
